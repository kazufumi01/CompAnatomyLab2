cmake_minimum_required(VERSION 3.22)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
project(CompTests VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${JUCE_PATH} ${CMAKE_BINARY_DIR}/JUCE)
foreach(T PTest DTest)
  juce_add_console_app(${T} PRODUCT_NAME ${T})
  target_include_directories(${T} PRIVATE ${PLUG}/Source)
  target_compile_definitions(${T} PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JucePlugin_Name="Test" JucePlugin_Build_Standalone=0 JUCE_MODAL_LOOPS_PERMITTED=1)
  target_link_libraries(${T} PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra)
endforeach()
target_sources(PTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/shot.cpp ${PLUG}/Source/PluginProcessor.cpp ${PLUG}/Source/PluginEditor.cpp)
target_sources(DTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/dsp.cpp ${PLUG}/Source/PluginProcessor.cpp ${PLUG}/Source/PluginEditor.cpp)
